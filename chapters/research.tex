\chapter{Preliminary Research}
\label{ch:research}

\section{Literature Review}
\label{sec:literature}
\guidance{Review the research literature relevant to the problem: the
established methods, their results, and their limitations. Synthesise the
findings rather than listing papers one by one, and state what the
literature establishes and what it leaves open.}

\section{Existing Solutions and Technologies}
\label{sec:related}
\guidance{Survey the products, systems, and technologies already available in
this space, including anything the industry partner already uses: what they
do, their strengths, and their limitations. Conclude with the gap this
project addresses. Candidate solutions for this specific problem are
formulated later, in \Cref{sec:alternatives}.}

\section{Market and User Research}
\label{sec:market}
\guidance{Identify who would use the solution and establish the demand for
it: the size and characteristics of the user base or market, what users
currently do instead, and what they need. Report the evidence gathered
(surveys, interviews, discussions with the industry partner, published
data). The financial implications are analysed in \Cref{sec:economics}.}

\section{Feasibility Study}
\label{sec:feasibility}
\guidance{Assess whether the project is feasible along the dimensions that
matter to it: technical (available methods, data, and computing resources),
economic (cost against expected benefit), operational (whether the intended
users and organisation can adopt it), and legal or regulatory. State the
conclusion and any assumption the project depends on.}
